```typescript
// 路由节点定义
interface RouteNode {
  path: string;              // 原始路径表达式
  method: string;            // HTTP方法
  filePath: string;          // 文件路径
  lineNumber: number;        // 行号
  scope: Scope;              // 作用域信息
  ast: ts.Node;              // AST节点
}

// 作用域信息
interface Scope {
  variables: Map<string, VariableInfo>;  // 局部变量
  imports: Map<string, ImportInfo>;      // 导入信息
  functions: Map<string, FunctionInfo>;  // 函数定义
  parent?: Scope;                        // 父作用域
}

// 变量信息
interface VariableInfo {
  name: string;
  value: string | ts.Node;
  type: string;
  isConst: boolean;
  definedAt: { file: string; line: number };
}
```

### 1.2 主推导流程

```typescript
class URLDeriver {
  private symbolTable: Map<string, SymbolInfo>;
  private fileCache: Map<string, ts.SourceFile>;
  
  /**
   * 推导完整URL
   * 这是整个URL推导的入口函数
   */
  async deriveURL(node: RouteNode): Promise<DerivationResult> {
    try {
      let url = node.path;
      const steps: DerivationStep[] = [];
      
      // Step 1: 检查是否是字面量（最简单情况）
      if (this.isStringLiteral(url)) {
        return {
          success: true,
          url: url,
          confidence: 1.0,
          steps: [{ type: 'literal', input: url, output: url }]
        };
      }
      
      // Step 2: 展开模板字符串
      if (this.isTemplateString(url)) {
        const expanded = await this.expandTemplateString(url, node.scope);
        steps.push({ type: 'template_expansion', input: url, output: expanded });
        url = expanded;
      }
      
      // Step 3: 内联函数调用
      if (this.hasFunctionCall(url)) {
        const inlined = await this.inlineFunctionCalls(url, node.scope);
        steps.push({ type: 'function_inline', input: url, output: inlined });
        url = inlined;
      }
      
      // Step 4: 处理三元表达式
      if (this.hasTernaryExpression(url)) {
        const resolved = this.resolveTernary(url, node.scope);
        steps.push({ type: 'ternary_resolution', input: url, output: resolved });
        url = resolved;
      }
      
      // Step 5: 解析局部变量
      if (this.hasVariableReference(url)) {
        const resolved = await this.resolveVariables(url, node.scope);
        steps.push({ type: 'variable_resolution', input: url, output: resolved });
        url = resolved;
      }
      
      // Step 6: 处理字符串拼接
      if (this.hasStringConcatenation(url)) {
        const concatenated = this.evaluateConcatenation(url, node.scope);
        steps.push({ type: 'concatenation', input: url, output: concatenated });
        url = concatenated;
      }
      
      // 计算置信度
      const confidence = this.calculateConfidence(steps);
      
      return {
        success: true,
        url: url,
        confidence: confidence,
        steps: steps
      };
      
    } catch (error) {
      return {
        success: false,
        url: node.path,
        confidence: 0,
        error: error.message,
        steps: []
      };
    }
  }
  
  /**
   * 展开模板字符串
   * 例如: `${API_PREFIX}/users` → `/api/v1/users`
   */
  private async expandTemplateString(
    template: string, 
    scope: Scope
  ): Promise<string> {
    // 匹配 ${...} 模式
    const regex = /\$\{([^}]+)\}/g;
    let result = template;
    const matches = Array.from(template.matchAll(regex));
    
    for (const match of matches) {
      const [fullMatch, expression] = match;
      
      // 评估表达式
      const value = await this.evaluateExpression(expression.trim(), scope);
      
      // 替换
      result = result.replace(fullMatch, value);
    }
    
    return result;
  }
  
  /**
   * 评估表达式
   * 支持：变量引用、属性访问、函数调用、三元表达式
   */
  private async evaluateExpression(
    expr: string, 
    scope: Scope
  ): Promise<string> {
    // 1. 简单变量引用
    if (this.isSimpleIdentifier(expr)) {
      return this.resolveVariable(expr, scope);
    }
    
    // 2. 属性访问 (例如: config.apiPrefix)
    if (expr.includes('.')) {
      return this.resolvePropertyAccess(expr, scope);
    }
    
    // 3. 函数调用 (例如: getPrefix())
    if (expr.includes('(')) {
      return this.resolveFunctionCall(expr, scope);
    }
    
    // 4. 三元表达式 (例如: isDev ? '/dev' : '/prod')
    if (expr.includes('?')) {
      return this.resolveTernary(expr, scope);
    }
    
    // 5. 字符串拼接 (例如: 'api' + version)
    if (expr.includes('+')) {
      return this.evaluateConcatenation(expr, scope);
    }
    
    // 6. 字面量
    return expr.replace(/['"]/g, '');
  }
  
  /**
   * 解析变量引用
   * 支持跨文件追踪
   */
  private resolveVariable(name: string, scope: Scope): string {
    // 1. 在当前作用域查找
    let currentScope: Scope | undefined = scope;
    while (currentScope) {
      const variable = currentScope.variables.get(name);
      if (variable) {
        if (typeof variable.value === 'string') {
          return variable.value;
        } else {
          // 如果是AST节点，需要进一步解析
          return this.evaluateASTNode(variable.value, scope);
        }
      }
      currentScope = currentScope.parent;
    }
    
    // 2. 在导入中查找
    const importInfo = scope.imports.get(name);
    if (importInfo) {
      return this.resolveImportedVariable(importInfo);
    }
    
    // 3. 在全局符号表查找
    const symbol = this.symbolTable.get(name);
    if (symbol) {
      return symbol.value;
    }
    
    // 4. 未找到，返回原始名称
    return `{${name}}`;
  }
  
  /**
   * 解析导入的变量（跨文件追踪）
   */
  private async resolveImportedVariable(
    importInfo: ImportInfo
  ): Promise<string> {
    const { modulePath, exportName } = importInfo;
    
    // 1. 读取目标文件
    const sourceFile = await this.getSourceFile(modulePath);
    
    // 2. 查找导出的变量
    const exportedVar = this.findExportedVariable(sourceFile, exportName);
    
    if (!exportedVar) {
      return `{${exportName}}`;
    }
    
    // 3. 解析变量值
    return this.evaluateASTNode(exportedVar.initializer, this.buildScope(sourceFile));
  }
  
  /**
   * 内联函数调用
   * 只内联简单的纯函数
   */
  private async inlineFunctionCalls(
    expr: string, 
    scope: Scope
  ): Promise<string> {
    const callRegex = /(\w+)\(([^)]*)\)/g;
    let result = expr;
    
    for (const match of expr.matchAll(callRegex)) {
      const [fullMatch, funcName, argsStr] = match;
      
      // 1. 查找函数定义
      const funcDef = this.findFunctionDefinition(funcName, scope);
      if (!funcDef) {
        continue; // 无法找到定义，跳过
      }
      
      // 2. 检查是否可以内联（必须是纯函数）
      if (!this.canInline(funcDef)) {
        continue;
      }
      
      // 3. 解析参数
      const args = this.parseArguments(argsStr, scope);
      
      // 4. 内联函数体
      const inlinedValue = this.inlineFunction(funcDef, args);
      
      // 5. 替换
      result = result.replace(fullMatch, inlinedValue);
    }
    
    return result;
  }
  
  /**
   * 检查函数是否可以内联
   * 条件：
   * 1. 函数体只有一个return语句
   * 2. 没有副作用（不修改外部状态）
   * 3. 不包含复杂逻辑（循环、条件等）
   */
  private canInline(funcDef: FunctionInfo): boolean {
    const body = funcDef.body;
    
    // 检查是否只有一个return语句
    if (body.statements.length !== 1) {
      return false;
    }
    
    const statement = body.statements[0];
    if (!ts.isReturnStatement(statement)) {
      return false;
    }
    
    // 检查return表达式的复杂度
    const returnExpr = statement.expression;
    if (!returnExpr) {
      return false;
    }
    
    // 不允许包含：函数调用、循环、条件语句
    const hasComplexity = this.hasComplexNode(returnExpr);
    
    return !hasComplexity;
  }
  
  /**
   * 内联函数
   * 将参数替换到函数体中
   */
  private inlineFunction(
    funcDef: FunctionInfo, 
    args: string[]
  ): string {
    const returnStatement = funcDef.body.statements[0] as ts.ReturnStatement;
    const returnExpr = returnStatement.expression!;
    
    // 构建参数映射
    const paramMap = new Map<string, string>();
    funcDef.parameters.forEach((param, index) => {
      paramMap.set(param.name, args[index] || '');
    });
    
    // 替换参数
    let result = returnExpr.getText();
    for (const [paramName, argValue] of paramMap) {
      const regex = new RegExp(`\\b${paramName}\\b`, 'g');
      result = result.replace(regex, argValue);
    }
    
    return result;
  }
  
  /**
   * 处理三元表达式
   * 例如: isDev ? '/dev' : '/prod'
   * 
   * 使用状态机解析，处理嵌套情况
   */
  private resolveTernary(expr: string, scope: Scope): string {
    // 状态机状态
    enum State {
      CONDITION,
      TRUE_BRANCH,
      FALSE_BRANCH
    }
    
    let state = State.CONDITION;
    let condition = '';
    let trueBranch = '';
    let falseBranch = '';
    let depth = 0; // 括号深度
    
    for (let i = 0; i < expr.length; i++) {
      const char = expr[i];
      
      // 跟踪括号深度
      if (char === '(') depth++;
      if (char === ')') depth--;
      
      switch (state) {
        case State.CONDITION:
          if (char === '?' && depth === 0) {
            state = State.TRUE_BRANCH;
          } else {
            condition += char;
          }
          break;
          
        case State.TRUE_BRANCH:
          if (char === ':' && depth === 0) {
            state = State.FALSE_BRANCH;
          } else {
            trueBranch += char;
          }
          break;
          
        case State.FALSE_BRANCH:
          falseBranch += char;
          break;
      }
    }
    
    // 评估条件
    const conditionResult = this.evaluateCondition(condition.trim(), scope);
    
    // 返回对应分支
    const selectedBranch = conditionResult ? trueBranch : falseBranch;
    return this.evaluateExpression(selectedBranch.trim(), scope);
  }
  
  /**
   * 评估条件表达式
   */
  private evaluateCondition(condition: string, scope: Scope): boolean {
    // 1. 解析变量
    const resolved = this.resolveVariable(condition, scope);
    
    // 2. 评估布尔值
    if (resolved === 'true') return true;
    if (resolved === 'false') return false;
    
    // 3. 检查环境变量
    if (condition.includes('process.env')) {
      return this.evaluateEnvCondition(condition);
    }
    
    // 4. 默认返回false（保守策略）
    return false;
  }
  
  /**
   * 计算推导置信度
   * 基于推导步骤的复杂度
   */
  private calculateConfidence(steps: DerivationStep[]): number {
    let confidence = 1.0;
    
    for (const step of steps) {
      switch (step.type) {
        case 'literal':
          confidence *= 1.0; // 字面量，100%确定
          break;
        case 'template_expansion':
          confidence *= 0.95; // 模板展开，95%确定
          break;
        case 'variable_resolution':
          confidence *= 0.9; // 变量解析，90%确定
          break;
        case 'function_inline':
          confidence *= 0.85; // 函数内联，85%确定
          break;
        case 'ternary_resolution':
          confidence *= 0.8; // 三元表达式，80%确定
          break;
        default:
          confidence *= 0.7; // 其他情况，70%确定
      }
    }
    
    return confidence;
  }
}
```

```typescript
class MultiStrategySearchEngine {
  private strategies: Map<string, SearchStrategy>;
  private cache: SearchCache;
  
  constructor(
    private db: DatabaseService,
    private vectorSearch: VectorSearchService,
    private dependencyGraph: DependencyGraph
  ) {
    this.initializeStrategies();
  }
  
  /**
   * 主搜索入口
   * 融合多种策略的结果
   */
  async search(
    query: string, 
    options: SearchOptions = {}
  ): Promise<SearchResult[]> {
    // 1. 检查缓存
    const cacheKey = this.buildCacheKey(query, options);
    const cached = this.cache.get(cacheKey);
    if (cached) {
      return cached;
    }
    
    // 2. 选择策略
    const strategies = this.selectStrategies(query, options);
    
    // 3. 并行执行所有策略
    const strategyResults = await Promise.all(
      strategies.map(async strategy => {
        const startTime = Date.now();
        try {
          const results = await strategy.search(query, options);
          const latency = Date.now() - startTime;
          
          // 记录性能指标
          this.recordMetrics(strategy.name, latency, results.length);
          
          return { strategy: strategy.name, results, latency };
        } catch (error) {
          console.error(`Strategy ${strategy.name} failed:`, error);
          return { strategy: strategy.name, results: [], latency: 0 };
        }
      })
    );
    
    // 4. 结果融合
    const merged = this.mergeResults(strategyResults, options);
    
    // 5. 去重
    const deduped = this.deduplicateResults(merged);
    
    // 6. 重排序
    const ranked = this.rankResults(deduped, query, options);
    
    // 7. 截取Top N
    const final = ranked.slice(0, options.limit || 10);
    
    // 8. 缓存结果
    this.cache.set(cacheKey, final);
    
    return final;
  }
  
  /**
   * 选择搜索策略
   * 根据查询特征自动选择
   */
  private selectStrategies(
    query: string, 
    options: SearchOptions
  ): SearchStrategy[] {
    const selected: SearchStrategy[] = [];
    
    // 策略1: 向量搜索（总是启用）
    selected.push(this.strategies.get('vector')!);
    
    // 策略2: 精确匹配（如果查询像是完整路径）
    if (this.looksLikeExactPath(query)) {
      selected.push(this.strategies.get('exact')!);
    }
    
    // 策略3: 模糊匹配（如果查询包含通配符或部分路径）
    if (this.needsFuzzyMatch(query)) {
      selected.push(this.strategies.get('fuzzy')!);
    }
    
    // 策略4: 依赖感知（如果明确要求查找相关API）
    if (options.includeDependencies || this.queryMentionsDependencies(query)) {
      selected.push(this.strategies.get('dependency')!);
    }
    
    // 策略5: 图搜索（如果查询涉及调用链）
    if (this.queryMentionsCallChain(query)) {
      selected.push(this.strategies.get('graph')!);
    }
    
    return selected;
  }
  
  /**
   * 结果融合算法
   * 使用加权平均 + 位置加权
   */
  private mergeResults(
    strategyResults: StrategyResult[],
    options: SearchOptions
  ): SearchResult[] {
    // 策略权重配置
    const weights = options.weights || {
      vector: 0.4,
      exact: 0.3,
      fuzzy: 0.15,
      dependency: 0.1,
      graph: 0.05
    };
    
    // 位置权重：排名越靠前，权重越高
    const positionWeight = (rank: number) => 1 / Math.log2(rank + 2);
    
    // 构建结果映射
    const resultMap = new Map<string, MergedResult>();
    
    for (const { strategy, results } of strategyResults) {
      const strategyWeight = weights[strategy] || 0;
      
      results.forEach((result, index) => {
        const key = `${result.url}:${result.method}:${result.filePath}`;
        
        if (!resultMap.has(key)) {
          resultMap.set(key, {
            ...result,
            scores: {},
            finalScore: 0
          });
        }
        
        const merged = resultMap.get(key)!;
        
        // 计算该策略的贡献分数
        const posWeight = positionWeight(index);
        const contribution = result.score * strategyWeight * posWeight;
        
        merged.scores[strategy] = contribution;
        merged.finalScore += contribution;
      });
    }
    
    return Array.from(resultMap.values());
  }
  
  /**
   * 去重算法
   * 基于URL + 文件路径
   */
  private deduplicateResults(results: SearchResult[]): SearchResult[] {
    const seen = new Set<string>();
    const deduped: SearchResult[] = [];
    
    for (const result of results) {
      const key = `${result.url}:${result.filePath}`;
      
      if (!seen.has(key)) {
        seen.add(key);
        deduped.push(result);
      } else {
        // 如果已存在，取分数更高的
        const existingIndex = deduped.findIndex(
          r => `${r.url}:${r.filePath}` === key
        );
        if (result.finalScore > deduped[existingIndex].finalScore) {
          deduped[existingIndex] = result;
        }
      }
    }
    
    return deduped;
  }
  
  /**
   * 重排序算法
   * 综合考虑多个因子
   */
  private rankResults(
    results: SearchResult[],
    query: string,
    options: SearchOptions
  ): SearchResult[] {
    return results.sort((a, b) => {
      // 因子1: 融合分数（权重50%）
      const scoreDiff = (b.finalScore - a.finalScore) * 0.5;
      
      // 因子2: 查询匹配度（权重20%）
      const matchA = this.calculateQueryMatch(query, a);
      const matchB = this.calculateQueryMatch(query, b);
      const matchDiff = (matchB - matchA) * 0.2;
      
      // 因子3: 最近修改时间（权重15%）
      const recencyA = this.calculateRecency(a.updatedAt);
      const recencyB = this.calculateRecency(b.updatedAt);
      const recencyDiff = (recencyB - recencyA) * 0.15;
      
      // 因子4: 使用频率（权重15%）
      const popularityA = this.getPopularity(a.url);
      const popularityB = this.getPopularity(b.url);
      const popularityDiff = (popularityB - popularityA) * 0.15;
      
      const totalDiff = scoreDiff + matchDiff + recencyDiff + popularityDiff;
      
      return totalDiff > 0 ? -1 : 1;
    });
  }
}
```

```typescript
class VectorSearchStrategy implements SearchStrategy {
  name = 'vector';
  
  constructor(
    private db: DatabaseService,
    private embedding: EmbeddingService
  ) {}
  
  async search(query: string, options: SearchOptions): Promise<SearchResult[]> {
    // 1. 生成查询向量
    const queryVector = await this.embedding.embed(query);
    
    // 2. 向量相似度搜索
    const results = await this.db.query(`
      SELECT 
        id,
        url,
        method,
        file_path,
        line_number,
        description,
        1 - (embedding <=> $1) as score
      FROM routes
      WHERE 1 - (embedding <=> $1) > $2
      ${options.method ? 'AND method = $3' : ''}
      ORDER BY embedding <=> $1
      LIMIT $4
    `, [
      queryVector,
      options.threshold || 0.7,
      options.method,
      options.limit || 20
    ]);
    
    return results.rows.map(row => ({
      id: row.id,
      url: row.url,
      method: row.method,
      filePath: row.file_path,
      lineNumber: row.line_number,
      description: row.description,
      score: row.score
    }));
  }
}
```

```typescript
class AgentReasoningEngine {
  private maxRounds = 5;
  private context: ConversationContext;
  
  async reason(userQuery: string): Promise<AgentResponse> {
    this.context.reset();
    this.context.addUserMessage(userQuery);
    
    for (let round = 0; round < this.maxRounds; round++) {
      // 构建prompt
      const prompt = this.buildPrompt(round);
      
      // 调用LLM
      const response = await this.llm.complete(prompt, {
        tools: this.tools.getAvailableTools(),
        temperature: 0.3
      });
      
      // 处理工具调用
      if (response.toolCalls) {
        await this.handleToolCalls(response.toolCalls);
        continue;
      }
      
      // 检查质量
      const quality = await this.assessQuality(response.content);
      if (quality.score >= 0.75) {
        return this.buildResponse(response, round + 1);
      }
      
      // 添加反思反馈
      this.context.addReflection(quality.feedback);
    }
    
    throw new Error('Max rounds exceeded');
  }
  
  private buildPrompt(round: number): string {
    return `
你是一个代码搜索助手。

用户问题：${this.context.getUserQuery()}

${this.context.getToolResults()}

${this.context.getReflections()}

请回答用户问题，必要时使用工具获取信息。
    `;
  }
}
```

```typescript
  const client = await pool.connect();

  } catch (error) {
    await client.query('ROLLBACK');
    throw error;
  } finally {
    client.release();
  }
}
```
